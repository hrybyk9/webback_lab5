Перевірка проксі-сервера за допомогою curl (Git Bash)
Сервер запущено командою:
  node main.js -h 127.0.0.1 -p 3000 -c ./cache

=== Перевірка опцій ===
# немає обов'язкової опції -> помилка, ненульовий код виходу
node main.js -p 3000 -c ./cache
echo $?

=== GET ===
# 1. Картинки ще немає в кеші -> береться з http.cat (X-Image-Source: http.cat)
curl -i -o cat200.jpg http://127.0.0.1:3000/200
# 2. Повторний запит -> з кешу (X-Image-Source: cache)
curl -s -D - -o /dev/null http://127.0.0.1:3000/200
# 3. Неіснуючий код -> 404
curl -i http://127.0.0.1:3000/999
# 4. Неправильний шлях -> 400
curl -i http://127.0.0.1:3000/abc

=== PUT ===
# 5. Записати власну картинку my.jpg для коду 200 -> 201
curl -i -X PUT --data-binary @my.jpg -H "Content-Type: image/jpeg" http://127.0.0.1:3000/200
# 6. GET після PUT -> повертається моя картинка (з кешу)
curl -s -D - -o after_put.jpg http://127.0.0.1:3000/200
cmp my.jpg after_put.jpg && echo "Файли однакові"
# 7. PUT з порожнім тілом -> 400
curl -i -X PUT http://127.0.0.1:3000/201
